fun int main()
    var int x, y;
    x = 1;
    while x < 10 do
        y = 1;
        while y < 10 do
            if (x + y) == 8 then
                break
            endif;
            y = y + 1
        endwhile;
        if x == 6 then
            break
        endif;
        print(x * 100 + y);
        x = x + 1
    endwhile;
    return(0)
endfun
